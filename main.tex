\begin{frame}[fragile]{문제 소개}{Problem - 코드 단독 슬라이드}
  % 한글 주석도 사용 가능합니다 (코드 폰트: DM Mono / 한글: Noto Sans)
  \begin{minted}{cpp}
    #include <bits/stdc++.h>
    using namespace std;

    int main() {
        ios::sync_with_stdio(0); cin.tie(0);  // 빠른 입출력
        int N, M;
        cin >> N >> M;
        vector<long long> P(N + 1);
        for (int i = 1; i <= N; i++) {
            int x; cin >> x;
            P[i] = P[i - 1] + x;              // 누적 합 전처리
        }
        while (M--) {
            int L, R; cin >> L >> R;
            cout << P[R] - P[L - 1] << '\n';
        }
    }
  \end{minted}
\end{frame}
