\begin{frame}{그리디 알고리즘}{<Greedy Algorithm>}
  \concept{%
    \hl{미래를 고려하지 않고, 매 순간 최선의 선택을 쌓아가며}\\
    결과적으로 최적의 답을 구해내는 알고리즘}
  \vspace{4mm}
  \annot{단, 현재의 선택이 미래의 최적해 가능성을 배제하는 경우 사용할 수 없어요}
\end{frame}

\begin{frame}{그리디 알고리즘}{<Greedy Algorithm> - 구현 방법}
  \begin{enumerate}
    \item 그리디하게 접근할 수 있는 문제인지 확인한다.
    \begin{itemize}
      \item[] \annot{그리디로 모든 문제를 풀 수 있는 건 아니에요. 문제를 부분 문제들로 나누어
        풀어갈 수 있는 경우, 그리고 현재 단계에서의 최선의 선택이 전체의 최선의 선택으로
        이어지는 경우에만 사용 가능해요.}
    \end{itemize}
    \item 이 문제의 매 단계마다 할 수 있는 최선의 선택을 찾고, 예외 케이스가 있는지 살핀다.
    \begin{itemize}
      \item[] \annot{현재의 선택이 다음 선택에 영향을 준다면 어떤 선택이 최선인지 알기 힘들어요.
        당시에 최선으로 보이던 선택을 나중에 다시 바꾸어야 할 경우에는, 최선의 그리디한
        전략이 아니에요.}
    \end{itemize}
    \item 정답에 도달할 때까지 2번을 반복한다.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{그리디 알고리즘}{<Greedy Algorithm> - 연습 문제}
  \begin{columns}[T]
    \column{0.40\textwidth}
      \vspace{2mm}
      \problemtitle{4}{거스름돈}{5585}
      \vspace{8mm}

      \subhead{코드 설명}
      \begin{itemize}
        \item 가치가 큰 것부터 교환하면 될 거 같고, 실제로 그렇게 구현하면 됩니다.
      \end{itemize}
    \column{0.60\textwidth}
      \begin{minted}{cpp}
        int main(){
            int cost;
            cin >> cost;
            int change = 1000 - cost;
            int count = 0;
            // 사용할 수 있는 동전 단위 (큰 순서대로)
            int coins[6] = {500, 100, 50, 10, 5, 1};
            // 매 단계에서 가장 가치가 높은 동전부터 '그리디하게' 선택
            for (int i = 0; i < 6; i++) {
                if (change == 0) break;   // 거스름돈을 다 줬다면 종료
                // 현재 동전으로 줄 수 있는 최대 개수를 구함
                count += (change / coins[i]);
                // 남은 거스름돈 업데이트 (나머지 연산)
                change %= coins[i];
            }
            cout << count << endl;   // 결과 출력
        }
      \end{minted}
  \end{columns}
\end{frame}

    \item 만약 거스름돈이 14원이라면, $(14 / 5 = 2)$ 5원짜리 동전 2개와 2원짜리 동전 2개를
      사용하면 된다.
    \item[] \vspace{1mm}
    \item 그런데 만약, \alert{거스름돈이 1원이라면? 3원이라면?}

\begin{frame}{그리디 알고리즘}{<Greedy Algorithm> - 구현 방법}
  \begin{enumerate}
    \item 그리디하게 풀 수 있는 문제인지 확인한다.
    \item 이 문제의 매 단계마다 할 수 있는 최선의 선택을 찾고, 예외 케이스가 있는지 살핀다.
    \item 정답에 다다를 때까지 2번을 반복한다.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{그리디 알고리즘}{<Greedy Algorithm> - 구현 코드}
  \begin{columns}[T]
    \column{0.40\textwidth}
      \vspace{2mm}
      \problemtitle{6}{거스름돈}{14916}
      \vspace{8mm}

      \begin{itemize}
        \item \hl{예외 케이스는 따로!}
        \item[] \vspace{1mm}
        \item \hl{그리디 알고리즘은 앞에서부터 / 뒤에서부터 순차적으로!}
      \end{itemize}
    \column{0.60\textwidth}
      \begin{minted}{cpp}
        int main(){
            int charge, answer = 0;
            cin >> charge;
            // 예외 처리
            if(charge == 1 || charge == 3){
                cout << -1 << '\n';
            }
            else{
                // 5로 나누어 떨어질 때까지 2씩 빼주며 카운트
                while(charge > 0){
                    if(charge % 5 == 0){ // 5로 나누어 떨어질 때
                        answer += charge / 5;
                        break;
                    }
                    else{ // 5로 나누어 떨어지지 않을 때
                        charge -= 2;
                        answer++;
                    }
                }
                cout << answer << '\n';
            }
        }
      \end{minted}
  \end{columns}
\end{frame}
