  \subhead{문제 요약}
  \begin{itemize}
    \item \alert{로프 N개}가 있다. 각각의 로프는 들어올릴 수 있는 최대 중량 제한이 다르다.
    \item K개의 로프를 사용하여 중량이 w인 물체를 들어올리면, 각 로프에는 \alert{w/K}만큼의 중량이 걸린다.
    \item 로프를 마음껏 사용할 때, 들어올릴 수 있는 \alert{최대 중량}은?
  \end{itemize}
\end{frame}

\begin{frame}{그리디 알고리즘}{<Greedy Algorithm>}
  \begin{center}
    \problemtitle{7}{로프}{2217}
  \end{center}
  \vspace{4mm}
  \subhead{문제 풀이}
  \begin{itemize}
    \item 로프를 \alert{x개} 사용한다고 가정합시다.
    \item 중량 제한이 낮은 로프를 사용하는 것보다는 높은 로프를 사용하는 것이 이득입니다.
    \item 따라서, \alert{중량 제한이 가장 높은 로프} x개를 골라야 합니다.
    \item 로프를 1개, 2개, …, N개 고르는 경우를 모두 시도해 보아야 합니다.
    \item 중량 제한이 높은 순으로 정렬한 뒤, 1개, 2개, …, N개 고르는 경우를 모두 시도합시다. \\ \alert{(사용하는 로프 중량의 최솟값)*(사용하는 로프 개수)의 최대}가 정답이 됩니다.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{그리디 알고리즘}{<Greedy Algorithm>}
  \begin{minted}{cpp}
    int main(){
      int N;
      cin >> N;
      vector<int> A(N);
      for (int i = 0; i < N; i++)
        cin >> A[i];
      sort(A.rbegin(), A.rend());
      int ans = 0;
      for (int i = 0; i < N; i++)
        ans = max(ans, A[i] * (i + 1));
      
      cout << ans;
    }
  \end{minted}
\end{frame}
