    \begin{itemize}
      \item \texttt{isPrime()} 함수의 연산 횟수는 약 $\sqrt{A}$번
      \item[] \vspace{1mm}
      \item 주어진 자연수를 $A$라 하면,
      \item 연산 횟수 $T(N) = \sum \sqrt{A}\, \le\, N\sqrt{A}$
      \item $A$의 최댓값은 1,000이므로, $T(N) = \sqrt{1000}\,N$
    \end{itemize}
  \colsplit
    \begin{minted}[fontsize=\footnotesize]{cpp}
      #include <iostream>
      using namespace std;
      int isPrime(int A) {
          if(A == 1) return 0;
          for(int i=2; i*i <= A; i++){
              if(A % i == 0) return 0;
          }
          return 1;
      }
      int main() {
          int N, ans=0;
          cin >> N;
          for(int i=0; i < N; i++){
              int x; cin >> x;
              ans += isPrime(x);
          }
          cout << ans;
      }
    \end{minted}
  \end{twocol}

\begin{frame}{시간 복잡도}{<Time Complexity> - 점근 표기법}
  함수에서 가장 중요한 정보: \alert{최고차항의} \hl{차수}
  \vspace{3mm}
  \begin{itemize}
    \item $f(x) = 3x^2 + 10,\ g(x) = 100x$
    \item[] \hspace{2em}%
      \begin{tabular}{@{}ll@{}}
        – $f(1) = 13$          & $g(1) = 100$      \\
        – $f(10) = 310$        & $g(10) = 1{,}000$  \\
        – $f(100) = 30{,}010$  & $g(100) = 10{,}000$ \\
        – $f(1000) = 3{,}000{,}010$ & $g(1000) = 100{,}000$ \\
      \end{tabular}
    \item[] \vspace{2mm}
    \item $x$가 \alert{특정 수보다 크면,} \hl{항상 $f(x)>g(x)$를 만족함}
  \end{itemize}
\end{frame}

\begin{frame}{시간 복잡도}{<Time Complexity> - 점근 표기법}
  시간 복잡도는 대개 Big-O 표기법을 사용해서 표기합니다.
  \vspace{3mm}
  \begin{itemize}
    \item $f(n) \in O\left(g(n)\right)$
    \begin{itemize}
      \item 어떤 양수 $n_0$와 $c$가 존재하여
      \item $n \ge n_0$을 만족하는 모든 $n$에 대해
      \item $0 \le f(n) \le cg(n)$을 만족한다.
    \end{itemize}
    \item[] \vspace{2mm}
    \item ..?
  \end{itemize}
\end{frame}

\begin{frame}{시간 복잡도}{<Time Complexity> - 점근 표기법}
  \begin{itemize}
    \item $f(n) \in O\left(g(n)\right)$
    \begin{itemize}
      \item 어떤 양수 $n_0$와 $c$가 존재하여 $n \ge n_0$을 만족하는 모든 $n$에 대해
      \item $0 \le f(n) \le cg(n)$을 만족한다.
    \end{itemize}
    \item[] \vspace{3mm}
    \item $n \ge n_0$을 만족하는 모든 $n$
    \begin{itemize}
      \item $n$이 \alert{한 없이 커지면} 항상 부등호 성립
    \end{itemize}
    \item 어떤 양수 $c$
    \begin{itemize}
      \item 최고차항 \alert{계수 무시} \& 최고차항의 \alert{차수만 확인}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{시간 복잡도}{<Time Complexity> - 점근 표기법}
  \begin{enumerate}
    \item $f(n) = 2n+3,\ g(n) = n$
    \begin{itemize}
      \item $n_0 = 3,\, c = 3$이면, $2n+3 \le 3n$ : $2n+3 \in O(n)$
    \end{itemize}
    \item[] \vspace{2mm}
    \item $f(n) = n^2+4n+3,\ g(n) = n^2$
    \begin{itemize}
      \item $n_0 = 5,\, c = 2$이면, $n^2+4n+3 \le 2n^2$ : $n^2+4n+3 \in O(n^2)$
    \end{itemize}
  \end{enumerate}
  \vspace{9mm}
  \annot{Big-O Notation의 의미는 $f(x)$가 아무리 빨리 증가해도,\\
    ``최대'' $g(x)$에 비례하는 수준으로 증가한다는 것입니다.\\
    따라서 $f(n)=n+1$ 역시 $O(n^2)$이라 쓸 수 있어요!}
\end{frame}

\begin{frame}{시간 복잡도}{<Time Complexity> - 왜 중요한가?}
  \begin{itemize}
    \item 간단한 연산($+$, $=$ 등)은 1초에 10억 번 정도 실행되고,\\
      복잡한 연산(\%, \texttt{sqrt()} 등)은 1초에 1억 번 정도 실행됨
    \begin{itemize}
      \item Online Judge 시스템 기준으로, 사용 환경에 따라 크게 다를 수 있음
      \item 대부분의 상황에선 \hl{연산 1억 번 = 1초} 라고 기억해도 일반적으로 무방함
    \end{itemize}
    \item[] \vspace{3mm}
    \item 문제 조건이 주어질 때, 주어진 시간 안에 프로그램이 실행되려면\\
      어떤 시간 복잡도가 필요한지 가늠할 수 있음
    \begin{itemize}
      \item 문제 조건이 $N=10$만이라면? \complexity{N}이나 \complexity{N \log N}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{시간 복잡도}{<Time Complexity>}
  $O(1) < O(\log N) < O\left(\sqrt{N}\right) < O(N) < O(N \log N)
    < O(N^2) < \cdots < O(2^N) < O(N!)$
