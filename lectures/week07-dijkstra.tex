\begin{frame}[fragile]{다익스트라 알고리즘}{<Dijkstra's Algorithm> - 구현 코드}
  \begin{twocol}
    \begin{enumerate}
      \item 시작 노드로부터 다른 노드 사이의 최단 거리를 모두 INF로 설정한다.
      \item 모든 노드에 방문할 때까지 다음을 반복한다.
      \begin{enumerate}
        \item 방문하지 않은 노드 중 시작 노드와의 거리가 가장 짧은 노드에 방문한다.
        \item 방문한 노드와 인접한 노드의 최단 거리를 갱신한다.
        \item 선택한 노드를 방문 완료 상태로 바꾼다.
      \end{enumerate}
    \end{enumerate}
  \colsplit
    \begin{minted}[fontsize=\scriptsize]{cpp}
      // vector<int> dist(V+1, INF), K: 시작 정점 번호
      void dijkstra(vector<vector<pair<int,int>>>& G,
                    vector<int>& dist, int K) {
          priority_queue<pair<int, int>> PQ;
          PQ.push({0, K});
          dist[K] = 0;
          while(!PQ.empty()){
              int cost = -PQ.top().first;
              int curr = PQ.top().second;
              PQ.pop();
              for(int i=0; i < G[curr].size(); i++){
                  int next = G[curr][i].first;
                  int weight = G[curr][i].second;
                  if(dist[next] > cost + weight){
                      dist[next] = cost + weight;
                      PQ.push({-dist[next], next});
                  }
              }
          }
      }
    \end{minted}
  \end{twocol}
\end{frame}

\begin{frame}{다익스트라 알고리즘}{<Dijkstra's Algorithm> - 정당성 증명}
  \vspace{6mm}
  \begin{itemize}
    \item 다익스트라 알고리즘은 항상 가장 가까운 정점을 먼저 처리하므로, 이미 처리된
      정점까지의 거리는 최단거리임이 보장됩니다.
    \item 간선의 가중치가 모두 0 이상이기 때문에, 나중에 더 짧은 경로가 생길 수 없기
      때문입니다.
    \item[] \vspace{1mm}
    \item 따라서 매번 가장 짧은 경로를 확정하며 나아가는 방식이 \hl{전체 그래프에 대해
      올바른 최단거리를 보장}합니다.
  \end{itemize}
\end{frame}
